cmake_minimum_required(VERSION 3.22)

# Define project name and language types
project(stm32h5-firmware C ASM)

set(CMAKE_EXECUTABLE_SUFFIX ".elf")

# ----------------------------------------------------------------------
# 📌 1. Compile Definitions & Hardware Configurations
# ----------------------------------------------------------------------
# Change STM32H563xx to your exact chip variant (e.g., STM32H573xx, STM32H503xx)
add_compile_definitions(
    STM32H563xx
    USE_HAL_DRIVER
)

# ----------------------------------------------------------------------
# 📂 2. Include Paths (Header File Locations)
# ----------------------------------------------------------------------
include_directories(
    Core/Inc
    Drivers/STM32H5xx_HAL_Driver/Inc
    Drivers/STM32H5xx_HAL_Driver/Inc/Legacy
    Drivers/CMSIS/Device/ST/STM32H5xx/Include
    Drivers/CMSIS/Include
)

# ----------------------------------------------------------------------
# 📄 3. Source Files Gathering
# ----------------------------------------------------------------------
# Recursively look up files or explicitly list them
file(GLOB_RECURSE SOURCES
    "Core/Src/*.c"
    "Drivers/STM32H5xx_HAL_Driver/Src/*.c"
)

# Add your specific startup assembly file (Generated by CubeMX)
list(APPEND SOURCES
    Core/Startup/startup_stm32h563xx.s
)

# ----------------------------------------------------------------------
# 🎯 4. Executable target and Linker Configurations
# ----------------------------------------------------------------------
add_executable(${PROJECT_NAME} ${SOURCES})

# Target your exact Linker Script file path
set(LINKER_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/STM32H563ZITX_FLASH.ld")

target_link_options(${PROJECT_NAME} PRIVATE
    "-T${LINKER_SCRIPT}"
    "-mcpu=cortex-m33"
    "-mthumb"
    "-mfloat-abi=hard"
    "-mfpu=fpv5-sp-d16"
    "-Wl,-Map=${PROJECT_NAME}.map"
    "--specs=nano.specs"
    "-Wl,--gc-sections"
    "-Wl,--start-group"
    "-lc"
    "-lm"
    "-lnosys"
    "-Wl,--end-group"
)

# ----------------------------------------------------------------------
# 📦 5. Post-Build Commands (Generate Memory footprints, Hex & Bin files)
# ----------------------------------------------------------------------
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_SIZE} ${PROJECT_NAME}.elf
    COMMAND ${CMAKE_OBJCOPY} -O ihex ${PROJECT_NAME}.elf ${PROJECT_NAME}.hex
    COMMAND ${CMAKE_OBJCOPY} -O binary ${PROJECT_NAME}.elf ${PROJECT_NAME}.bin
    COMMENT "Generating size report, .hex, and .bin formats..."
)
