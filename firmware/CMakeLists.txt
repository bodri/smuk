cmake_minimum_required(VERSION 3.22)

# Define project name and language types
project(smuk C ASM)

set(CMAKE_EXECUTABLE_SUFFIX ".elf")

# ----------------------------------------------------------------------
# 📌 1. Compile Definitions & Hardware Configurations
# ----------------------------------------------------------------------
add_compile_definitions(
    DEBUG
    STM32H503xx
    USE_HAL_DRIVER
    USE_NUCLEO_64
)

# ----------------------------------------------------------------------
# 📂 2. Include Paths (Header File Locations)
# ----------------------------------------------------------------------
include_directories(
    Core/Inc
    Drivers/STM32H5xx_HAL_Driver/Inc
    Drivers/STM32H5xx_HAL_Driver/Inc/Legacy
    Drivers/CMSIS/Device/ST/STM32H5xx/Include
    Drivers/CMSIS/Include
    Drivers/BSP/STM32H5xx_Nucleo
    Drivers/SMU
    Platform/STM32H503
    Storage
    App
)

# ----------------------------------------------------------------------
# 📄 3. Source Files Gathering
# ----------------------------------------------------------------------
# Recursively look up files or explicitly list them
file(GLOB_RECURSE SOURCES
    "Core/Src/*.c"
    "Drivers/STM32H5xx_HAL_Driver/Src/*.c"
    "Drivers/BSP/STM32H5xx_Nucleo/*.c"
    "Drivers/SMU/*.c"
    "Platform/STM32H503/*.c"
    "Storage/*.c"
)

# Add your specific startup assembly file (Generated by CubeMX)
list(APPEND SOURCES
    Core/Startup/startup_stm32h503rbtx.s
)

# ----------------------------------------------------------------------
# 🎯 4. Executable target and Linker Configurations
# ----------------------------------------------------------------------
add_executable(${PROJECT_NAME} ${SOURCES})

target_compile_options(${PROJECT_NAME} PRIVATE
    "-mcpu=cortex-m33"
    "-mthumb"
    "-mfloat-abi=hard"
    "-mfpu=fpv5-sp-d16"
    "-O0"
    "-g3"
    "-Wall"
)

# Target your exact Linker Script file path
set(LINKER_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/STM32H503RBTX_FLASH.ld")

target_link_options(${PROJECT_NAME} PRIVATE
    "-T${LINKER_SCRIPT}"
    "-mcpu=cortex-m33"
    "-mthumb"
    "-mfloat-abi=hard"
    "-mfpu=fpv5-sp-d16"
    "-Wl,-Map=${PROJECT_NAME}.map"
    "--specs=nano.specs"
    "-Wl,--gc-sections"
    "-Wl,--start-group"
    "-lc"
    "-lm"
    # REMOVED "-lnosys" to prevent duplicate definition of '_exit' with syscalls.c
    "-Wl,--end-group"
)

# ----------------------------------------------------------------------
# 📦 5. Post-Build Commands (Generate Memory footprints, Hex & Bin files)
# ----------------------------------------------------------------------
# Explicitly force tool lookups so the variables are never empty strings
set(HEX_PROGRAM "arm-none-eabi-objcopy")
set(SIZE_PROGRAM "arm-none-eabi-size")

add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${SIZE_PROGRAM} ${PROJECT_NAME}.elf
    COMMAND ${HEX_PROGRAM} -O ihex ${PROJECT_NAME}.elf ${PROJECT_NAME}.hex
    COMMAND ${HEX_PROGRAM} -O binary ${PROJECT_NAME}.elf ${PROJECT_NAME}.bin
    COMMENT "Generating size report, .hex, and .bin formats..."
)
